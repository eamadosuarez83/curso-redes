% Ajustes de estilo para export_pdf.sh (se inyecta con pandoc -H).
%
% Figuras: pandoc (desde 3.5) escala cada imagen con \pandocbounded para que quepa en el
% alto de la página, pero no descuenta el pie de figura, y las muy altas (el
% mapa del curso) se salen. Aquí el límite es el 85 % del alto del texto.
% Las figuras pequeñas conservan su tamaño natural. Con pandoc anterior a la
% 3.5 (que no define \pandocbounded) el ajuste simplemente no se aplica.
\makeatletter
\ifdefined\pandocbounded
\renewcommand*\pandocbounded[1]{%
  \sbox\pandoc@box{#1}%
  \Gscale@div\@tempa{0.85\textheight}{\dimexpr\ht\pandoc@box+\dp\pandoc@box\relax}%
  \Gscale@div\@tempb{\linewidth}{\wd\pandoc@box}%
  \ifdim\@tempb\p@<\@tempa\p@\let\@tempa\@tempb\fi
  \ifdim\@tempa\p@<\p@\scalebox{\@tempa}{\usebox\pandoc@box}%
  \else\usebox{\pandoc@box}%
  \fi
}
\fi
\makeatother

% Código: las salidas de comandos de red tienen líneas largas (ip link, ss…)
% que se salían del margen derecho. fvextra las parte y marca la
% continuación con una flecha. Aplica a los bloques con lenguaje
% (```console, entorno Highlighting) y, al redefinir verbatim, también a los
% bloques sin lenguaje.
\usepackage{fvextra}
\fvset{breaklines=true, breakanywhere=true}
\makeatletter
\@ifundefined{Highlighting}{}{%
  \RecustomVerbatimEnvironment{Highlighting}{Verbatim}{commandchars=\\\{\},breaklines,breakanywhere}}
\makeatother
\RecustomVerbatimEnvironment{verbatim}{Verbatim}{}
